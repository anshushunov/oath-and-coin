> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Закрыть Gate 0 — компилируемый контрактный пакет, детерминированный simulation spike и CI, после чего владелец продукта может дать ручное подтверждение по `MVP_PLAN` §4.3.

**Architecture:** `simulation/` — чистая .NET-библиотека без единой ссылки на Godot; она держит авторитетное состояние, детерминированный RNG, конверт событий и causal trace. `tools/` — консольные приложения без движка: валидатор контента и раннер сценариев. Godot-проекта в этом плане нет вовсе: Gate 0 не содержит UI-задач, поэтому весь код проверяется быстрым контуром `dotnet test`.

**Tech Stack:** .NET 8.0.424, C# 12, xUnit, `JsonSchema.Net`, `System.Text.Json`, GitHub Actions.

## Global Constraints

Эти требования входят в acceptance criteria **каждой** задачи.

- **Нет ссылок на Godot** нигде в `simulation/`, `tools/`, `tests/`. Godot-проект в Gate 0 не создаётся.
- **Запрещены в `simulation/`:** `DateTime.Now`, `DateTime.UtcNow`, `Guid.NewGuid()`, `System.Random`, `Environment.TickCount`, чтение файловой системы, зависимость от `CultureInfo.CurrentCulture`. Основание: `TDD` §7.3.
- **Целочисленная арифметика** во всех игровых расчётах. `float`/`double` в `simulation/` не используются. Основание: `TDD` §7.4.
- **Детерминированный порядок:** ни одна итерация по `Dictionary`/`HashSet` не влияет на результат. Где нужен порядок — явная сортировка по `ContentId`. Основание: `TDD` §5.1.
- **`TreatWarningsAsErrors`** включён; сборка с предупреждением считается упавшей.
- **`Nullable` enabled**, `LangVersion 12.0`, `Deterministic true`.
- **Версия SDK пинуется** через `global.json` значением `8.0.424` — это единственный SDK на машине разработки (`dotnet --list-sdks`).
- **Формат commit-сообщений:** Conventional Commits (`feat:`, `fix:`, `docs:`, `chore:`, `test:`, `ci:`). Авторство LLM в сообщении не указывается.
- **Каждое число в PR сопровождается командой, которой оно снято.** Основание: спека §8.1.
- **Каждая новая проверка сопровождается мутантом**, показывающим, что она краснеет. Основание: спека §8.2.

**Спека:** `docs/superpowers/specs/2026-08-12-technical-rules-design.md`

**Вне scope этого плана:** runtime harness (отдельный план перед M1, спека §7), Godot-проект, UI, бой, экономика, сохранения, портреты героев.

```csharp
namespace OathAndCoin.Simulation.Random;

/// <summary>
/// PCG32 по референсной реализации pcg_basic.c. Единственный источник
/// случайности в simulation core: System.Random запрещён, потому что не
/// гарантирует одинаковую последовательность между версиями .NET (TDD §7.1).
/// </summary>
public sealed class Pcg32
{
    private const ulong Multiplier = 6364136223846793005UL;

    private ulong _state;
    private readonly ulong _increment;

    public Pcg32(ulong seed, ulong sequence)
    {
        _increment = (sequence << 1) | 1UL;
        _state = 0UL;
        NextUInt32();
        _state = unchecked(_state + seed);
        NextUInt32();
    }

    public uint NextUInt32()
    {
        ulong previous = _state;
        _state = unchecked(previous * Multiplier + _increment);

        uint xorShifted = (uint)(((previous >> 18) ^ previous) >> 27);
        int rotation = (int)(previous >> 59);

        return (xorShifted >> rotation) | (xorShifted << ((-rotation) & 31));
    }

    /// <summary>Равномерное целое в [minInclusive, maxExclusive) без смещения.</summary>
    public int NextInt32(int minInclusive, int maxExclusive)
    {
        if (maxExclusive <= minInclusive)
        {
            throw new ArgumentOutOfRangeException(
                nameof(maxExclusive),
                $"maxExclusive ({maxExclusive}) must be greater than minInclusive ({minInclusive}).");
        }

        uint span = (uint)(maxExclusive - minInclusive);
        uint threshold = (uint)((0x1_0000_0000UL - span) % span);

        while (true)
        {
            uint draw = NextUInt32();
            if (draw >= threshold)
            {
                return minInclusive + (int)(draw % span);
            }
        }
    }
}
```

```csharp
namespace OathAndCoin.Simulation.Random;

/// <summary>
/// Derivation потоков из campaign seed: номер потока становится sequence
/// PCG32, поэтому потоки независимы и воспроизводимы по одному seed.
/// Косметический поток никогда не меняет симуляцию (TDD §7.2).
/// </summary>
public sealed class RngStreams
{
    private readonly Dictionary<RngStream, Pcg32> _streams;

    public RngStreams(ulong campaignSeed)
    {
        CampaignSeed = campaignSeed;
        _streams = Enum.GetValues<RngStream>()
            .ToDictionary(stream => stream, stream => new Pcg32(campaignSeed, (ulong)stream));
    }

    public ulong CampaignSeed { get; }

    public Pcg32 Get(RngStream stream) => _streams[stream];
}
```

`simulation/OathAndCoin.Simulation/Ids/ContentId.cs`:

```csharp
using System.Text.RegularExpressions;

namespace OathAndCoin.Simulation.Ids;

/// <summary>
/// Стабильный namespaced идентификатор контента вида "namespace:name".
/// Не зависит от отображаемого имени и не переиспользуется (TDD §5.1).
/// Конструируется только через Parse/TryParse, поэтому невалидного
/// значения в системе не существует.
/// </summary>
public readonly struct ContentId : IEquatable<ContentId>, IComparable<ContentId>
{
    private static readonly Regex SegmentPattern = new(
        "^[a-z][a-z0-9_]*$",
        RegexOptions.Compiled | RegexOptions.CultureInvariant);

    private readonly string? _namespace;
    private readonly string? _name;

    private ContentId(string @namespace, string name)
    {
        _namespace = @namespace;
        _name = name;
    }

    public string Namespace => _namespace ?? throw Uninitialized();

    public string Name => _name ?? throw Uninitialized();

    public string Value => $"{Namespace}:{Name}";

    public static bool TryParse(string? value, out ContentId id)
    {
        id = default;

        if (string.IsNullOrEmpty(value))
        {
            return false;
        }

        string[] segments = value.Split(':');
        if (segments.Length != 2)
        {
            return false;
        }

        if (!SegmentPattern.IsMatch(segments[0]) || !SegmentPattern.IsMatch(segments[1]))
        {
            return false;
        }

        id = new ContentId(segments[0], segments[1]);
        return true;
    }

    public static ContentId Parse(string? value) => TryParse(value, out ContentId id)
        ? id
        : throw new FormatException(
            $"Invalid content id: '{value}'. Expected 'namespace:name' where both segments match ^[a-z][a-z0-9_]*$.");

    public bool Equals(ContentId other) =>
        string.Equals(_namespace, other._namespace, StringComparison.Ordinal)
        && string.Equals(_name, other._name, StringComparison.Ordinal);

    public override bool Equals(object? obj) => obj is ContentId other && Equals(other);

    public override int GetHashCode() => HashCode.Combine(_namespace, _name);

    public int CompareTo(ContentId other) => string.CompareOrdinal(Value, other.Value);

    public override string ToString() => Value;

    public static bool operator ==(ContentId left, ContentId right) => left.Equals(right);

    public static bool operator !=(ContentId left, ContentId right) => !left.Equals(right);

    public static bool operator <(ContentId left, ContentId right) => left.CompareTo(right) < 0;

    public static bool operator >(ContentId left, ContentId right) => left.CompareTo(right) > 0;

    public static bool operator <=(ContentId left, ContentId right) => left.CompareTo(right) <= 0;

    public static bool operator >=(ContentId left, ContentId right) => left.CompareTo(right) >= 0;

    private static InvalidOperationException Uninitialized() =>
        new("ContentId was not initialised. Use ContentId.Parse or ContentId.TryParse.");
}
```

```csharp
namespace OathAndCoin.Simulation.Decisions;

/// <summary>Один фактор решения: код причины, источник и величина влияния.</summary>
public sealed record TraceFactor(string ReasonCode, string SourceEntity, int Magnitude);

/// <summary>
/// Машинно-читаемое объяснение решения (TDD §8). Строится тем же вычислением,
/// что и само решение, а не реконструируется постфактум.
/// </summary>
public sealed record CausalTrace
{
    public required long TraceId { get; init; }
    public required IReadOnlyList<TraceFactor> PositiveFactors { get; init; }
    public required IReadOnlyList<TraceFactor> NegativeFactors { get; init; }
    public required IReadOnlyList<string> BlockedBy { get; init; }
    public required string? TieBreak { get; init; }
}

/// <summary>Результат автономного решения вместе с его объяснением.</summary>
public sealed record DecisionResult
{
    private readonly string _selectedAction = string.Empty;

    public required string SelectedAction
    {
        get => _selectedAction;
        init
        {
            _selectedAction = value;
            EnsureSelectionIsConsidered();
        }
    }

    private readonly IReadOnlyList<string> _consideredActions = [];

    public required IReadOnlyList<string> ConsideredActions
    {
        get => _consideredActions;
        init
        {
            _consideredActions = value;
            EnsureSelectionIsConsidered();
        }
    }

    public required int SelectedScore { get; init; }

    public required CausalTrace Trace { get; init; }

    private void EnsureSelectionIsConsidered()
    {
        if (string.IsNullOrEmpty(_selectedAction) || ConsideredActions.Count == 0)
        {
            return;
        }

        if (!ConsideredActions.Contains(_selectedAction, StringComparer.Ordinal))
        {
            throw new ArgumentException(
                $"Selected action '{_selectedAction}' is not among considered actions " +
                $"[{string.Join(", ", ConsideredActions)}]. TDD §8 invariant.",
                nameof(SelectedAction));
        }
    }
}
```

> **Почему проверка вызывается из обоих сеттеров.** Порядок инициализации в object initializer не гарантирован: `ConsideredActions` может быть присвоен как до, так и после `SelectedAction`. Поэтому `EnsureSelectionIsConsidered` вызывается из обоих `init` и молча выходит, пока вторая половина данных не задана. Оба свойства используют явные backing fields, а не ключевое слово `field`: последнее на `LangVersion 12.0` недоступно.

```csharp
using OathAndCoin.Simulation.Events;
using OathAndCoin.Simulation.Ids;

namespace OathAndCoin.Simulation.State;

/// <summary>
/// Метаданные воспроизводимости (TDD §7.1, §12). Реального времени здесь нет:
/// оно может быть метаданными файла сохранения, но на симуляцию не влияет.
/// </summary>
public sealed record GameMetadata
{
    public required int SaveSchemaVersion { get; init; }
    public required string RulesetVersion { get; init; }
    public required string ContentVersion { get; init; }
    public required ulong CampaignSeed { get; init; }
    public required long LogicalTime { get; init; }
    public required long NextEventId { get; init; }
}

/// <summary>
/// Авторитетное состояние кампании. Иммутабельно: переход состояния всегда
/// создаёт новый экземпляр, поэтому команду можно применить либо целиком,
/// либо не применить вовсе (TDD §6.1).
/// </summary>
public sealed record GameState
{
    public required GameMetadata Metadata { get; init; }
    public required IReadOnlyDictionary<HeroId, HeroState> Heroes { get; init; }
    public required IReadOnlyDictionary<ContentId, ContractState> Contracts { get; init; }
    public required IReadOnlyList<DomainEvent> History { get; init; }

    public HeroState Hero(HeroId id) => Heroes.TryGetValue(id, out HeroState? hero)
        ? hero
        : throw new KeyNotFoundException($"Unknown hero {id}.");

    public ContractState Contract(ContentId id) => Contracts.TryGetValue(id, out ContractState? contract)
        ? contract
        : throw new KeyNotFoundException($"Unknown contract {id}.");

    public GameState WithEvent(DomainEvent domainEvent)
    {
        if (domainEvent.EventId != Metadata.NextEventId)
        {
            throw new ArgumentException(
                $"Event id {domainEvent.EventId} does not match the expected next id {Metadata.NextEventId}. " +
                "Event ids come from the state counter, never from a clock or a random source.",
                nameof(domainEvent));
        }

        return this with
        {
            History = [.. History, domainEvent],
            Metadata = Metadata with { NextEventId = Metadata.NextEventId + 1 },
        };
    }
}
```

`simulation/OathAndCoin.Simulation/Content/ContentSet.cs`:

```csharp
using System.Text.Json;
using OathAndCoin.Simulation.Ids;
using OathAndCoin.Simulation.State;

namespace OathAndCoin.Simulation.Content;

/// <summary>
/// Загруженный и проверенный контент. Чтение файлов происходит здесь, на
/// границе core: дальше по системе ходят только уже проверенные значения.
/// </summary>
public sealed class ContentSet
{
    private static readonly JsonSerializerOptions SerializerOptions = new()
    {
        PropertyNameCaseInsensitive = false,
        ReadCommentHandling = JsonCommentHandling.Disallow,
        AllowTrailingCommas = false,
    };

    private ContentSet(
        IReadOnlyDictionary<ContentId, HeroDefinition> heroes,
        IReadOnlyDictionary<ContentId, ContractDefinition> contracts,
        string contentVersion)
    {
        Heroes = heroes;
        Contracts = contracts;
        ContentVersion = contentVersion;
    }

    public IReadOnlyDictionary<ContentId, HeroDefinition> Heroes { get; }

    public IReadOnlyDictionary<ContentId, ContractDefinition> Contracts { get; }

    public string ContentVersion { get; }

    public static ContentSet Load(string contentRoot)
    {
        Dictionary<ContentId, HeroDefinition> heroes = [];
        foreach (string file in EnumerateJson(Path.Combine(contentRoot, "heroes")))
        {
            HeroDocument document = Deserialize<HeroDocument>(file);
            ContentId id = ParseId(document.Id, file);
            HeroDefinition definition = new(
                id,
                document.DisplayNameKey,
                document.Greed,
                document.Caution,
                document.TrustInGuild);

            if (!heroes.TryAdd(id, definition))
            {
                throw new InvalidDataException($"Duplicate hero id '{id}' found in '{file}'.");
            }
        }

        Dictionary<ContentId, ContractDefinition> contracts = [];
        foreach (string file in EnumerateJson(Path.Combine(contentRoot, "contracts")))
        {
            ContractDocument document = Deserialize<ContractDocument>(file);
            ContentId id = ParseId(document.Id, file);
            ContractDefinition definition = new(
                id,
                document.DisplayNameKey,
                document.Payment,
                document.Risk);

            if (!contracts.TryAdd(id, definition))
            {
                throw new InvalidDataException($"Duplicate contract id '{id}' found in '{file}'.");
            }
        }

        if (heroes.Count == 0)
        {
            throw new InvalidDataException($"No hero definitions found under '{contentRoot}'.");
        }

        return new ContentSet(heroes, contracts, contentVersion: "0.1.0");
    }

    /// <summary>
    /// Собирает начальное состояние. HeroId выдаётся по возрастанию ContentId,
    /// а не по порядку файлов в каталоге: порядок перечисления файловой
    /// системы не должен влиять на симуляцию (TDD §5.1).
    /// </summary>
    public GameState CreateInitialState(ulong campaignSeed, string rulesetVersion)
    {
        Dictionary<HeroId, HeroState> heroes = [];
        int index = 0;

        foreach (HeroDefinition definition in Heroes.Values.OrderBy(hero => hero.Id))
        {
            HeroId id = new(index);
            heroes[id] = new HeroState
            {
                Id = id,
                Definition = definition.Id,
                DisplayNameKey = definition.DisplayNameKey,
                Greed = definition.Greed,
                Caution = definition.Caution,
                TrustInGuild = definition.TrustInGuild,
            };
            index++;
        }

        Dictionary<ContentId, ContractState> contracts = [];
        foreach (ContractDefinition definition in Contracts.Values.OrderBy(contract => contract.Id))
        {
            contracts[definition.Id] = new ContractState
            {
                Id = definition.Id,
                Payment = definition.Payment,
                Risk = definition.Risk,
                Status = ContractStatus.Offered,
            };
        }

        return new GameState
        {
            Metadata = new GameMetadata
            {
                SaveSchemaVersion = 1,
                RulesetVersion = rulesetVersion,
                ContentVersion = ContentVersion,
                CampaignSeed = campaignSeed,
                LogicalTime = 0,
                NextEventId = 1,
            },
            Heroes = heroes,
            Contracts = contracts,
            History = [],
        };
    }

    private static IEnumerable<string> EnumerateJson(string directory)
    {
        if (!Directory.Exists(directory))
        {
            throw new InvalidDataException($"Content directory '{directory}' does not exist.");
        }

        // Сортировка по ordinal делает перечисление независимым от файловой системы.
        return Directory.GetFiles(directory, "*.json").OrderBy(path => path, StringComparer.Ordinal);
    }

    private static T Deserialize<T>(string file)
    {
        string json = File.ReadAllText(file);

        try
        {
            return JsonSerializer.Deserialize<T>(json, SerializerOptions)
                ?? throw new InvalidDataException($"File '{file}' deserialised to null.");
        }
        catch (JsonException error)
        {
            throw new InvalidDataException($"File '{file}' is not valid JSON: {error.Message}", error);
        }
    }

    private static ContentId ParseId(string raw, string file) => ContentId.TryParse(raw, out ContentId id)
        ? id
        : throw new InvalidDataException($"File '{file}' declares an invalid id '{raw}'.");
}
```

```csharp
using Json.Schema;

            $"content ok: {content.Heroes.Count} heroes, {content.Contracts.Count} contracts, version {content.ContentVersion}");
    }
    catch (InvalidDataException error)
    {
        failures.Add(error.Message);
    }
}

foreach (string failure in failures)
{
    Console.Error.WriteLine($"content error: {failure}");
}

return failures.Count == 0 ? 0 : 1;

void ValidateDirectory(string directory, string schemaPath)
{
    if (!Directory.Exists(directory))
    {
        failures.Add($"missing content directory '{directory}'");
        return;
    }

    if (!File.Exists(schemaPath))
    {
        failures.Add($"missing schema '{schemaPath}'");
        return;
    }

    JsonSchema schema = JsonSchema.FromFile(schemaPath);

    foreach (string file in Directory.GetFiles(directory, "*.json").OrderBy(path => path, StringComparer.Ordinal))
    {
        JsonNode? node;
        try
        {
            node = JsonNode.Parse(File.ReadAllText(file));
        }
        catch (JsonException error)
        {
            failures.Add($"'{file}' is not valid JSON: {error.Message}");
            continue;
        }

        EvaluationResults result = schema.Evaluate(node, new EvaluationOptions
        {
            OutputFormat = OutputFormat.List,
        });

        if (!result.IsValid)
        {
            failures.Add($"'{file}' violates '{Path.GetFileName(schemaPath)}'");
        }
    }
}
```

`simulation/OathAndCoin.Simulation/Decisions/ContractDecisionRule.cs`:

```csharp
using OathAndCoin.Simulation.Random;
using OathAndCoin.Simulation.State;

namespace OathAndCoin.Simulation.Decisions;

/// <summary>
/// Правило решения о контракте для спайка Gate 0.
///
/// НАМЕРЕННО МИНИМАЛЬНО. Выбор между utility model и rule/priority model —
/// открытый вопрос BQ-004, решаемый в Milestone 1 (MVP_PLAN §5.4). Здесь
/// нужен только доказанный контракт: решение и trace возвращаются одним
/// вычислением, а не реконструируются постфактум (TDD §8).
///
/// Вся арифметика целочисленная (TDD §7.4).
/// </summary>
public static class ContractDecisionRule
{
    public const string Accept = "accept";
    public const string Decline = "decline";

    private const int JitterMinInclusive = -5;
    private const int JitterMaxExclusive = 6;

    public static DecisionResult Decide(HeroState hero, ContractState contract, Pcg32 rng, long traceId)
    {
        int paymentPull = contract.Payment * hero.Greed / 100;
        int riskPush = contract.Risk * hero.Caution / 100;
        int trustPull = hero.TrustInGuild / 10;
        int mood = rng.NextInt32(JitterMinInclusive, JitterMaxExclusive);

        int score = paymentPull - riskPush + trustPull + mood;

        List<TraceFactor> positive =
        [
            new(ReasonCodes.PaymentAttractive, contract.Id.Value, paymentPull),
            new(ReasonCodes.TrustsTheGuild, hero.Definition.Value, trustPull),
        ];

        List<TraceFactor> negative =
        [
            new(ReasonCodes.RiskTooHigh, contract.Id.Value, -riskPush),
        ];

        if (mood >= 0)
        {
            positive.Add(new TraceFactor(ReasonCodes.UnpredictableMood, hero.Definition.Value, mood));
        }
        else
        {
            negative.Add(new TraceFactor(ReasonCodes.UnpredictableMood, hero.Definition.Value, mood));
        }

        return new DecisionResult
        {
            ConsideredActions = [Accept, Decline],
            SelectedAction = score >= 0 ? Accept : Decline,
            SelectedScore = score,
            Trace = new CausalTrace
            {
                TraceId = traceId,
                PositiveFactors = positive,
                NegativeFactors = negative,
                BlockedBy = [],
                TieBreak = score == 0 ? "tie_break.non_negative_score_accepts" : null,
            },
        };
    }
}
```

```csharp
using OathAndCoin.Simulation.Commands;
using OathAndCoin.Simulation.Decisions;
using OathAndCoin.Simulation.Events;
using OathAndCoin.Simulation.Random;
using OathAndCoin.Simulation.State;

namespace OathAndCoin.Simulation;

/// <summary>
/// Применяет команды к состоянию. Валидация выполняется до любого изменения,
/// поэтому отклонённая команда возвращает исходный экземпляр состояния.
/// </summary>
public sealed class SimulationEngine
{
    private readonly RngStreams _streams;

    public SimulationEngine(RngStreams streams) => _streams = streams;

    public CommandResult Apply(GameState state, ProposeContractToHero command)
    {
        if (!state.Heroes.ContainsKey(command.HeroId))
        {
            return CommandResult.Rejected(state, "rejected.unknown_hero");
        }

        if (!state.Contracts.TryGetValue(command.ContractId, out ContractState? contract))
        {
            return CommandResult.Rejected(state, "rejected.unknown_contract");
        }

        if (contract.Status != ContractStatus.Offered)
        {
            return CommandResult.Rejected(state, "rejected.contract_already_resolved");
        }

        HeroState hero = state.Hero(command.HeroId);
        long traceId = state.Metadata.NextEventId;

        DecisionResult decision = ContractDecisionRule.Decide(
            hero,
            contract,
            _streams.Get(RngStream.HeroDecision),
            traceId);

        bool accepted = decision.SelectedAction == ContractDecisionRule.Accept;

        DomainEvent domainEvent = accepted
            ? new HeroAcceptedContract(
                state.Metadata.NextEventId,
                state.Metadata.LogicalTime,
                traceId,
                hero.Id,
                contract.Id)
            : new HeroDeclinedContract(
                state.Metadata.NextEventId,
                state.Metadata.LogicalTime,
                traceId,
                hero.Id,
                contract.Id);

        Dictionary<Ids.ContentId, ContractState> contracts = new(state.Contracts)
        {
            [contract.Id] = contract with
            {
                Status = accepted ? ContractStatus.Accepted : ContractStatus.Declined,
            },
        };

        GameState next = (state with { Contracts = contracts }).WithEvent(domainEvent);

        return new CommandResult
        {
            Applied = true,
            RejectionCode = null,
            State = next,
            Events = [domainEvent],
            Decision = decision,
        };
    }
}
```

```csharp
using System.Text.Json;
using System.Text.Json.Serialization;
using OathAndCoin.Simulation.Commands;
using OathAndCoin.Simulation.Content;
using OathAndCoin.Simulation.Ids;
using OathAndCoin.Simulation.Random;
using OathAndCoin.Simulation.State;

namespace OathAndCoin.Simulation.Scenarios;

public sealed record ScenarioStep(
    long CommandId,
    string Hero,
    string Contract,
    string Action,
    int Score,
    IReadOnlyList<string> PositiveReasons,
    IReadOnlyList<string> NegativeReasons);

public sealed record ScenarioReport(
    string RulesetVersion,
    string ContentVersion,
    ulong CampaignSeed,
    IReadOnlyList<ScenarioStep> Steps);

internal sealed record ScenarioCommandDocument
{
    [JsonPropertyName("command_id")]
    public long CommandId { get; init; }

    [JsonPropertyName("hero_index")]
    public int HeroIndex { get; init; }

    [JsonPropertyName("contract")]
    public string Contract { get; init; } = string.Empty;
}

internal sealed record ScenarioDocument
{
    [JsonPropertyName("commands")]
    public IReadOnlyList<ScenarioCommandDocument> Commands { get; init; } = [];
}

/// <summary>
/// Headless-прогон сценария (TDD §14). Не зависит ни от движка, ни от UI:
/// это обычный вызов библиотеки, который одинаково работает из теста и из CLI.
/// </summary>
public static class ScenarioRunner
{
    public static ScenarioReport Run(string contentRoot, string commandsFile, ulong seed)
    {
        ContentSet content = ContentSet.Load(contentRoot);
        GameState state = content.CreateInitialState(seed, rulesetVersion: "0.1.0");
        SimulationEngine engine = new(new RngStreams(seed));

        ScenarioDocument scenario =
            JsonSerializer.Deserialize<ScenarioDocument>(File.ReadAllText(commandsFile))
            ?? throw new InvalidDataException($"Scenario file '{commandsFile}' deserialised to null.");

        List<ScenarioStep> steps = [];

        foreach (ScenarioCommandDocument document in scenario.Commands)
        {
            HeroId heroId = new(document.HeroIndex);
            ContentId contractId = ContentId.Parse(document.Contract);

            CommandResult result = engine.Apply(
                state,
                new ProposeContractToHero(document.CommandId, heroId, contractId));

            steps.Add(new ScenarioStep(
                document.CommandId,
                state.Heroes.TryGetValue(heroId, out HeroState? hero) ? hero.Definition.Value : heroId.ToString(),
                contractId.Value,
                result.Applied ? result.Decision!.SelectedAction : result.RejectionCode!,
                result.Decision?.SelectedScore ?? 0,
                result.Decision?.Trace.PositiveFactors.Select(f => f.ReasonCode).ToArray() ?? [],
                result.Decision?.Trace.NegativeFactors.Select(f => f.ReasonCode).ToArray() ?? []));

            state = result.State;
        }

        return new ScenarioReport(
            state.Metadata.RulesetVersion,
            state.Metadata.ContentVersion,
            state.Metadata.CampaignSeed,
            steps);
    }
}
```

> `File.ReadAllText` здесь допустим: `ScenarioRunner` — адаптер на границе core, как и `ContentSet.Load`. Запрет файловой системы глобальных ограничений относится к правилам симуляции, а не к загрузке данных.

```csharp
if (args.Length < 4 || args[0] != "run-scenario")
```
